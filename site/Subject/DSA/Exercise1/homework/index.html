    Bài 1: Flowers

    Tóm tắt đề bài:

    Cho dãy \(n\) số \(a_1, a_2, \dots, a_n\) và số nguyên \(k\). Yêu cầu in ra một dãy các phần tử nằm trong dãy đã cho, sao cho số lượng phần tử khác nhau là nhiều nhất.

    Cách làm:

    • Ở bài này có nhiều cách làm, có thể rời rạc hóa dãy số và lưu vào mảng tĩnh, hoặc như các làm của mình là áp dụng thẳng CTDL Map vào để lưu tần suất xuất hiện của các số luôn.

    • Đặt \(cnt[x]\) là tần suất xuất hiện của số \(x\) trong dãy (cnt là ctdl map). Gọi \(m\) là số lượng phần tử phân biệt trong dãy, ta có 2 trường hợp:

    • \(k \le m\):

    Trong trường hợp này ta chỉ cần lần lượt in ra \(k\) chữ số phân biệt mà thôi.

    1. \(k > m\):

    Trường hợp này mình sẽ in ra \(m\) chữ số phân biệt, trong quá trình in ra \(m\) số đó thì mình giảm luôn giá trị tần suất xuất hiện trong \(cnt\) đi, sau đó mình chỉ cần duyệt lxại cnt (ctdl map) để in ra các phần tử còn lại (thêm \(k - m\) phần tử) cho đủ số lượng mà thôi.

    Code full:

    #include<bits/stdc++.h>
#define fi first
#define se second
using namespace std;
using ll = long long;
using ii = pair<int, int>;

const int MOD = 1e9 + 7;
const int MAX = 1e5 + 5;cin >> n;
    for (int i = 1; i <= n; i ++) cin >> a[i].fi >> a[i].se;
    sort(a + 1, a + n + 1);

    for (int i = 1; i <= n; i ++) {
        cout << a[i].fi << ' ' << a[i].se << '\n';
    }

#define TASK "FLOWERS"

int n, k;
int a[MAX];

int main() {
    ios::sync_with_stdio(0); cin.tie(0); cout.tie(0);
    if (fopen(TASK".inp", "r")) {
        freopen(TASK".inp", "r", stdin);
        freopen(TASK".out", "w", stdout);
    }

    cin >> n >> k;
    for (int i = 1; i <= n; i++) {
        cin >> a[i];
    }

    map<int, int> cnt;
    for (int i = 1; i <= n; i ++) {
        cnt[a[i]]++;
    }

    for (ii x : cnt) {
        if (k == 0) break;
        k --;

        cnt[x.fi] --;
        cout << x.fi << ' ';
    }

    for (ii x : cnt) {
        while (k > 0 && x.se > 0) {
            k --; x.se --;
            cout << x.fi << ' ';
        }
    }

    return 0;
}

    Bài 2: Point2d

    Đề bài:

    Cho \(n\) điểm có dạng \((x, y)\) trên mặt phẳng Oxy, Hãy sắp xếp và in ra các điểm tăng dần theo \(x\), nếu \(x\) bằng nhau thì sắp xếp các điểm giảm dần theo \(y\).

    Cách làm:

    • Ở bài trên, đơn giản ta chỉ cần lưu các điểm bằng kiểu dữ liệu pair, khi ta sử dụng hàm \(sort\), ta sẽ định nghĩa thêm một hàm so sánh 2 phần tử pair theo định nghĩa của đề bài để đưa hàm so sánh đó vào hàm sort, cuối cùng chỉ việc in chúng ra.

    Code full:

    #include<bits/stdc++.h>
#define fi first
#define se second
using namespace std;
using ll = long long;
using ii = pair<int, int>;

const int MOD = 1e9 + 7;
const int MAX = 1e6 + 5;

#define TASK "point2d"

int n;
ii a[MAX];

bool cmp(ii x, ii y) {
    return x.fi < y.fi || (x.fi == y.fi && x.se > y.se);
}

int main() {
    ios::sync_with_stdio(0); cin.tie(0); cout.tie(0);
    if (fopen(TASK".inp", "r")) {
        freopen(TASK".inp", "r", stdin);
        freopen(TASK".out", "w", stdout);
    }

    cin >> n;
    for (int i = 1; i <= n; i ++) cin >> a[i].fi >> a[i].se;
    sort(a + 1, a + n + 1, cmp);

    for (int i = 1; i <= n; i ++) {
        cout << a[i].fi << ' ' << a[i].se << '\n';
    }

    return 0;
}

    Bài 3: GIFT

    Tóm tắt đề bài:

    Cho dãy gồm \(n\) số \(a_1, a_2, \dots, a_n\) và số nguyên dương \(x\). Chọn 2 số khác nhau trong dãy sao cho tổng là lớn nhất và không vượt quá \(x\).

    Giải thích cách làm:

    • Ý tưởng là sắp xếp dãy cho tăng dần, sau đó gọi \(u\) và \(v\) là 2 vị trí của món quà (\(u \le v\)) có tổng giá trị là lớn nhất và không vượt quá \(x\), khi đó ta chỉ cần duyệt giá trị \(v\) từ \(1\) đến \(n\), tại mỗi giá trị \(v\) ta chặt nhị phân trên dãy \(a_1 \dots a_{v - 1}\) để tìm \(u\) sao cho \(a_u \le x - a_v\) (tức là \(a_u + a_v \le x\)).
    • Cuối cùng ta chỉ lấy giá trị tổng lớn nhất trong \(n\) trường hợp đó là được.

    Code full:

    #include<bits/stdc++.h>
#define fi first
#define se second
using namespace std;
using ll = long long;
using ii = pair<int, int>;

const int MOD = 1e9 + 7;
const int MAX = 1e5 + 5;

#define TASK "gifts"

int n, x;
int a[MAX];

int main() {
    ios::sync_with_stdio(0); cin.tie(0); cout.tie(0);
    if (fopen(TASK".inp", "r")) {
        freopen(TASK".inp", "r", stdin);
        freopen(TASK".out", "w", stdout);
    }

    cin >> n >> x;
    for (int i = 1; i <= n; i ++) cin >> a[i];

    int ans(0);

    sort(a + 1, a + 1 + n);
    for (int v = 1; v <= n; v ++) {
        int val = x - a[v];

        int u = upper_bound(a + 1, a + v, val) - a - 1;

        if (u > 0) ans = max(ans, a[u] + a[v]);
    }

    cout << ans;

    return 0;
}

    Bài 4: PasswordStrength

    Giải thích đề bài:

    • Bài này yêu cầu kiểm tra độ mạnh của mật khẩu được nhập vào:
    \[score = BaseScore + (Num\_Excess * Bonus\_Excess) + (Num\_Upper * Bonus\_Upper) + (Num\_Numbers * Bonus\_Numbers) + (Num\_Symbols * Bonus\_Symbols) + Bonus\_Combo + Bonus\_FlatLower + Bonus\_FlatNumber\]

    Trong đó:

    Biến Giá trị Điều kiện
    BaseScore 40 Mặc định ban đầu
    Bonus_Excess 3 Mỗi ký tự vượt quá độ dài 8
    Bonus_Upper 4 Mỗi chữ cái in hoa
    Bonus_Number 5 Mỗi chữ số
    Bonus_Symbols 5 Mỗi ký tự đặc biệt

    Các điểm cộng thêm:

    • Bonus_Combo:
    • +25 nếu mật khẩu có cả chữ hoa, ký tự đặc biệt và số.
    • +15 nếu mật khẩu chứa hai trong ba loại trên.

    • +0 nếu chỉ có một loại hoặc không có loại nào.

    • Bonus_FlatLower:

    • -15 nếu mật khẩu chỉ toàn chữ cái thường.

    • 0 nếu ngược lại.

    • Bonus_FlatNumber:

    • -35 nếu mật khẩu chỉ toàn chữ số.

    • 0 nếu ngược lại.

    • Number_Execess = Hiệu giữa độ dài mật khẩu và 8.

    • Number_Upper = Số lượng chữ cái in hoa trong mật khẩu.
    • Number_Numbers = Số lượng chữ số trong mật khẩu.
    • Number_Symbols = Số lượng ký tự đặc biệt trong mật khẩu.

    Quy tắc kiểm tra mật khẩu hợp lệ - Tập ký tự đặc biệt hợp lệ gồm:
    ! @ # $ % ^ & * ? _ ~ - Mật khẩu không hợp lệ nếu: - Độ dài nhỏ hơn 8 ký tự. - Chứa một trong các ký tự . \ / , và dấu cách.

    Phân loại mật khẩu theo điểm số - < 50 → Yeu - 50 ≤ score < 75 → Vua - 75 ≤ score < 100 → Manh - ≥ 100 → RatManh - Không hợp lệ → KhongHopLe

    Giải thích cách làm

    Bài này là một bài implement, làm y như gì bài toán nói, để hạn chế bug thì mình sử dụng OOP để quản lý các thuộc tính và các phương thức dưới đây:

    Code full

    #include<bits/stdc++.h>
#define fi first
#define se second
using namespace std;
using ll = long long;
using ii = pair<int, int>;

const int MOD = 1e9 + 7;
const int MAX = 1e5 + 5;

#define TASK "passwordstrength"

class PasswordStrength {
private:
    int baseScore = 40;
    int bonus_Excess = 3;
    int bonus_Upper = 4;
    int bonus_Number = 5;
    int bonus_Symbols = 5;

    string s;

    int Number_Execess() {
        return s.length() - 8;
    }

    int Number_Upper() {
        int cnt = 0;
        for (char c : s) {
            if ('A' <= c && c <= 'Z') cnt++;
        }
        return cnt;
    }

    int Number_Numbers() {
        int cnt = 0;
        for (char c : s) {
            if ('0' <= c && c <= '9') cnt++;
        }
        return cnt;
    }

    int Number_Symbols() {
        string symbols = "!@#$%^&*?_~";
        int cnt = 0;
        for (char c : s) {
            bool ok = false;
            for (char x : symbols) if (c == x) ok = true;
            if (ok) cnt++;
        }
        return cnt;
    }

    int Bonus_Combo() {
        int cnt = 0;
        if (Number_Upper() > 0) cnt++;
        if (Number_Numbers() > 0) cnt++;
        if (Number_Symbols() > 0) cnt++;

        if (cnt == 3) return 25;
        if (cnt == 2) return 15;
        return 0;
    }

    int Bonus_FlatLower() {
        if (Number_Upper() == 0 && Number_Symbols() == 0 && Number_Numbers() == 0) return -15;
        return 0;
    }

    int Bonus_FlatNumber() {
        if (Number_Numbers() == s.length()) return -35;
        return 0;
    }

    int Score() {
        return baseScore + Number_Execess() * bonus_Excess + 
                            Number_Upper() * bonus_Upper + 
                            Number_Numbers() * bonus_Number +
                            Number_Symbols() * bonus_Symbols + 
                            Bonus_Combo() + Bonus_FlatLower() + Bonus_FlatNumber();
    }

    bool Invalid() {
        if (s.length() < 8) return true;

        string symbols_invalid = ".\\/ ,";
        for (char c : s) {
            for (char x : symbols_invalid) if (x == c) return true;
        }   
        return false;
    }

public:
    void Input() {
        cin >> s;
    }

    string GetState() {
        if (Invalid()) return "KhongHopLe";
        int score = Score();
        if (score < 50) return "Yeu";
        if (score < 75) return "Vua";
        if (score < 100) return "Manh";
        return "RatManh";
    }
};

int main() {
    ios::sync_with_stdio(0); cin.tie(0); cout.tie(0);
    if (fopen(TASK".inp", "r")) {
        freopen(TASK".inp", "r", stdin);
        freopen(TASK".out", "w", stdout);
    }

    PasswordStrength P;
    P.Input();
    cout << P.GetState();

    return 0;
}

    Bài 5:

    Tóm tắt đề bài:

    Mã hóa string sau bằng mật mã Caesar với \(k\) kí tự.

    Lời giải:

    • Duyệt từng kí tự, mỗi kí tự đổi qua số nguyên rồi cộng \(k\) vào, sau đó mod đi \(26\) rồi đổi lại thành kiểu kí tự.

    Code full

    #include<bits/stdc++.h>
#define fi first
#define se second
using namespace std;
using ll = long long;
using ii = pair<int, int>;

const int MOD = 1e9 + 7;
const int MAX = 1e5 + 5;

#define TASK "CaesarCipher"

int main() {
    ios::sync_with_stdio(0); cin.tie(0); cout.tie(0);
    if (fopen(TASK".inp", "r")) {
        freopen(TASK".inp", "r", stdin);
        freopen(TASK".out", "w", stdout);
    }

    int k;
    string s;

    cin >> k;

    bool ok = false;
    while (cin >> s) {
        if (ok) cout << ' ';
        ok = true;
        for (char c : s) {
            if (c != ' ') c = (c - 'A' + k) % 26 + 'A';
            cout << c;
        }
    }
    return 0;
}

    Bài 6: ReversingEncryption

    Mô tả đề bài:

    Một chuỗi \(s\) (plaintext) có độ dài \(n\) có thể được mã hóa bằng thuật toán sau: + Duyệt trên tất cả các ước của n theo thứ tự giảm dần từ \(n\) đến \(1\).
    + Với mỗi ước số \(d\), đảo ngược chuỗi con \(s\) \([1 \dots d]\) (tức là chuỗi con bắt đầu ở vị trí \(1\) và kết thúc ở vị trí \(d\)).

    Ví dụ, thuật toán trên được áp dụng cho chuỗi s= "ATTN2020" dẫn đến những thay đổi sau: "0202NTTA" → "2020NTTA" → "0220NTTA" → "0220NTTA" (rõ ràng là thao tác dảo ngược cuối cùng không thay đổi chuỗi vì d = 1).

    Lời giải:

    Ta sẽ lật ngược lại cách encode của bài toán, encode là duyệt các ước số theo thứ tự giảm dần thì mình sẽ duyệt các ước theo thứ tự tăng dần, mỗi lần duyệt mình lại reverse lại chuỗi con giống như các encode thôi.

    • Độ phức tạp: \(O(n^2)\).

    Code full:

    #include<bits/stdc++.h>
#define fi first
#define se second
using namespace std;
using ll = long long;
using ii = pair<int, int>;

const int MOD = 1e9 + 7;
const int MAX = 1e5 + 5;

#define TASK ""

int main() {
    ios::sync_with_stdio(0); cin.tie(0); cout.tie(0);
    if (fopen(TASK".inp", "r")) {
        freopen(TASK".inp", "r", stdin);
        freopen(TASK".out", "w", stdout);
    }

    string s;
    cin >> s;

    for (int d = 2; d <= s.size(); d ++) if (s.size() % d == 0) {
        for (int i = 0, j = d - 1; i < j; i ++, j --) {
            swap(s[i], s[j]);
        }
    }

    cout << s;

    return 0;
}

    Bài 7: Messages

    Mô tả đề bài:

    Xâu \(S\) là nội dung của bức thư nếu như \(Sb\) là phần đầu của xâu \(S\), \(Se\) là phần đuôi của \(S\). Hai bức thư \(Sb\) và \(Se\) đảm bảo đầy đủ nội dung của \(S\). Tuy nhiên có thể một phần cuối của \(Sb\) có thể được viết lặp lại trong phần đầu của \(Se\). Cho hai xâu \(Sb\) và \(Se\), hãy xác định một xâu \(S\) có thể là nội dung của bức thư sao cho độ dài của xâu S là ngắn nhất.

    Lời giải:

    • Vì mỗi xâu có độ dài không quá \(500\), Do đó ta có thể duyệt \(d\) là độ dài của phần chung theo chiều giảm dần, sau đó ta kiểm tra xem \(d\) kí tự cuối của xâu \(Sb\) có giống với \(d\) kí tự đầu của xâu \(Se\) hay không.

    Code full:

    #include<bits/stdc++.h>
#define fi first
#define se second
using namespace std;
using ll = long long;
using ii = pair<int, int>;

const int MOD = 1e9 + 7;
const int MAX = 1e5 + 5;

#define TASK ""

int main() {
    ios::sync_with_stdio(0); cin.tie(0); cout.tie(0);
    if (fopen(TASK".inp", "r")) {
        freopen(TASK".inp", "r", stdin);
        freopen(TASK".out", "w", stdout);
    }

    string a, b; cin >> a >> b;

    for (int d = min(a.size(), b.size()); d >= 0; d--) {
        bool ok = true;
        for (int i = 0; i < d; i ++) {
            if (a[a.size() - d + i] != b[i]) {
                ok = false;
                break;
            }
        }
        if (ok) {
            cout << a.size() + b.size() - d << '\n';
            cout << a;
            for (int i = d; i < b.size(); i++) cout << b[i];
            return 0;
        }
    }
    return 0;
}

    Bài 8: Binary Search 1

    Mô tả đề bài:

    • Cho dãy gồm \(n\) số và số \(x\), nếu \(x\) tồn tại trong dãy thì in ra vị trí của giá trị \(x\) đồng thời in ra số lần duyệt qua các phần tử trên thuật toán binary search trong quá trình tìm giá trị \(x\) đó.

    Lời giải:

    • Đơn giản là chỉ cần cài đặt thuật toán binary search, sau đó đếm số lần vòng while lặp mà thôi.

    Code full:

    #include<bits/stdc++.h>
#define fi first
#define se second
using namespace std;
using ll = long long;
using ii = pair<int, int>;

const int MOD = 1e9 + 7;
const int MAX = 1e5 + 5;

#define TASK ""

int n;
int a[MAX];

int main() {
    ios::sync_with_stdio(0); cin.tie(0); cout.tie(0);
    if (fopen(TASK".inp", "r")) {
        freopen(TASK".inp", "r", stdin);
        freopen(TASK".out", "w", stdout);
    }

    cin >> n;
    for (int i = 0; i < n; i++) {
        cin >> a[i];
    }
    int x; cin >> x;

    int cnt(0);

    int l = 0, r = n - 1;
    while (l <= r) {
        cnt ++;
        int mid = (l + r) / 2;
        if (x < a[mid]) r = mid - 1;
        else if (x > a[mid]) l = mid + 1;
        else {
            cout << mid << '\n';
            cout << cnt << '\n';
            return 0;
        }
    }
    cout << -1;

    return 0;
}

    Bài 10: Linear Search 1

    Mô tả đề bài:

    • Yêu cầu: Cài đặt thuật toán tìm kiếm tuyến tính để tìm số nguyên x trong mảng N phần tử.
    • Đầu vào:
    • Số nguyên N (0 < N < 10000).
    • N số nguyên là phần tử của mảng.
    • Số nguyên x cần tìm.
    • Đầu ra:
    • Vị trí đầu tiên của x trong mảng (tính từ đầu đến cuối). Nếu không tìm thấy, in -1.
    • Số lần duyệt để tìm x.
    • Nếu tìm thấy x, in vị trí đầu tiên của x khi duyệt từ cuối về đầu. Nếu không tìm thấy, in -1.
    • Số lần duyệt để tìm x từ cuối về đầu.
    • Lưu ý: Nếu không tìm thấy x từ đầu đến cuối, không cần thực hiện tìm kiếm từ cuối về đầu.

    Lời giải:

    • Làm như bài yêu cầu.

    Code full:

    #include<bits/stdc++.h>
using namespace std;

const int MAX = 1e5 + 5;

int a[MAX], n, x;

int main() {
    cin >> n;
    for (int i = 1; i <= n; i ++) cin >> a[i];
    cin >> x;

    int pos1 = -1, pos2;
    for (int i = 1; i <= n; i ++) {
        if (a[i] == x) {
            if (pos1 == -1) pos1 = i;
            pos2 = i;
        }
    }

    if (pos1 == -1) {
        cout << -1;
        return 0;
    }

    pos2 = n - pos2 + 1;
    cout << pos1 - 1 << '\n' << pos1 << '\n';
    cout << pos2 - 1 << '\n' << pos2 << '\n';

    return 0;
}

    Bài 11: Linear search 2

    Mô tả đề bài:

    Cài đặt thuật toán tìm kiếm tuyến tính để tìm tất cả vị trí của x trong mảng N phần tử và đếm số lần duyệt qua các phần tử.

    Lời giải:

    • Duyệt qua các phần tử để tìm, sau đó lưu chúng vào vector, rồi in ra size, vị trí, và dựa vào vị trí để in ra số lượng phần tử đã duyệt qua.

    Code full:

    #include<bits/stdc++.h>
using namespace std;
using ll = long long;

const int MAX = 1e5 + 5;

ll a[MAX], n, x;

int main() {
    ios::sync_with_stdio(0); cin.tie(0); cout.tie(0);
    cin >> n;
    for (int i = 1; i <= n; i ++) cin >> a[i];
    cin >> x;

    vector<int> ans;
    for (int i = 1; i <= n; i ++) {
        if (a[i] == x) {
            ans.push_back(i);
        }
    }

    cout << ans.size() << '\n';

    for (int i = 0; i < ans.size(); i ++) {
        cout << ans[i] - 1 << ' ' << ans[i] << '\n';
    }

    return 0;
}

    Bài 12: Linear search 3

    Mô tả đề bài:

    Cho trước một mảng có N phần tử. Hãy in ra \(MEX_i\) với mọi \(i\). \(MEX_i\) là MEX của dãy các phần tử từ \(a_0\) đến \(a_i\).

    Lời giải:

    Tạo một mảng \(cnt[]\) để lưu tần suất của các phần tử trong khi duyệt các phần tử tăng dần. Cùng lúc đó, duy trì biến \(ans\) là phần tử nhỏ nhất có tần suất là \(0\).

    Code full:

    #include<bits/stdc++.h>
using namespace std;
using ll = long long;

const int MAX = 2e5 + 5;

ll cnt[MAX], n;

int main() {
    ios::sync_with_stdio(0); cin.tie(0); cout.tie(0);
    cin >> n;

    int ans(0);

    for (int i = 1; i <= n; i ++) {
        int num; cin >> num;
        cnt[num] ++;

        while (cnt[ans] > 0) ans ++;

        cout << ans << " \n"[i == n];
    }
}

    Bài 13: Linear Search 5

    Mô tả đề bài:

    Cho một mảng A có độ dài N. Bạn phải chia các phần tử của A thành hai dãy con S1 và S2 (một trong S1, S2 có thể trống) sao cho mỗi phần tử của mảng A chỉ thuộc một trong hai dãy S1 hoặc S2. Tổng trọng số của hai dãy S1 và S2 là lớn nhất. Bạn in ra đáp án lớn nhất có thể của tổng trọng số của hai dãy S1 và S2.

    Lời giải:

    Nếu đưa ra nhận xét tốt thì ta dễ dàng nhận thấy kết quả là 2 số lớn nhất trừ 2 số bé nhất (xét riêng ra 2 trường hợp ngoại lệ \(n \le 3\)).

    Code full:

    #include<bits/stdc++.h>
using namespace std;
using ll = long long;

int main() {
    ios::sync_with_stdio(0); cin.tie(0); cout.tie(0);

    int t; cin >> t;

    while (t --) {
        int n; cin >> n;
        ll nummax1(0), nummax2(0), nummin1(1e9), nummin2(1e9);
        for (int i = 1; i <= n; i ++) {
            ll num; cin >> num;

            if (num >= nummax1) {
                nummax2 = nummax1;
                nummax1 = num;
            } else if (num > nummax2) nummax2 = num;

            if (num <= nummin1) {
                nummin2 = nummin1;
                nummin1 = num;
            } else if (num < nummin2) nummin2 = num;
        }

        if (n <= 3) cout << nummax1 - nummin1 << '\n';
        else cout << nummax1 + nummax2 - nummin1 - nummin2 << '\n';
    }
}

    Bài 14:

    Mô tả đề bài:

    Cho bảng \(m \times n\), in ra tổng các phần tử trong bảng con \(3\times 3\) là nhỏ nhất.

    Lời giải:

    • Đơn giản chỉ cần duyệt qua tất cả các bảng con \(3 \times 3\) để tìm giá trị bé nhất, còn về phàn kích thước của mảng đơn thuần ta chỉ cần sử dụng vector để lưu trữ kích thước mảng trên.

    Code full:

    #include<bits/stdc++.h>
using namespace std;
using ll = long long;

const int MAX = 1e6 + 7;

int n, m;
vector<int> a[MAX];

int main() {
    ios::sync_with_stdio(0); cin.tie(0); cout.tie(0);

    cin >> n >> m;
    for (int i = 0; i < n; i ++) {
        for (int j = 0; j < m; j ++) {
            int x; cin >> x;
            a[i].push_back(x);
        }
    }

    int ans(1e9);

    for (int i = 1; i < n - 1; i ++) {
        for (int j = 1; j < m - 1; j ++) {
            int sum(0);
            for (int u = i - 1; u <= i + 1; u ++)
                for (int v = j - 1; v <= j + 1; v ++)
                    sum += a[u][v];
            ans = min(ans, sum);
        }
    }

    cout << ans;
}
